cmake_minimum_required(VERSION 3.22)
project(OrvoEngine VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(JUCE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.deps/JUCE" CACHE PATH "JUCE source checkout")
if(NOT EXISTS "${JUCE_SOURCE_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Set JUCE_SOURCE_DIR to a JUCE 8 checkout")
endif()
add_subdirectory("${JUCE_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/juce" EXCLUDE_FROM_ALL)
juce_add_console_app(orvo-engine PRODUCT_NAME "orvo-engine")
target_sources(orvo-engine PRIVATE Main.cpp)
target_compile_definitions(orvo-engine PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_USE_MP3AUDIOFORMAT=1 JUCE_ASIO=1 JUCE_PLUGINHOST_VST3=1 JUCE_PLUGINHOST_AU=0 JUCE_PLUGINHOST_VST=0 JUCE_MODAL_LOOPS_PERMITTED=1 NOMINMAX)
if(MSVC)
  target_compile_options(orvo-engine PRIVATE /utf-8)
endif()
target_link_libraries(orvo-engine PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_recommended_config_flags)
enable_testing()
add_test(NAME engine-offline-self-check COMMAND orvo-engine --self-test)
